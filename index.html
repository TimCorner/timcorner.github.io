        // Normalise les positions (1, 2, 3...) et garantit la compatibilité avec une liste JSONBin existante
        function normalizeGiftOrders() {
            if (!Array.isArray(gifts)) gifts = [];
            gifts.forEach((g, idx) => {
                if (!g.id) g.id = 'gift-' + (Date.now() + idx);
                if (!Array.isArray(g.images) || g.images.length === 0) {
                    g.images = g.image ? [g.image] : ["https://placehold.co/600x400/e2e8f0/475569?text=Cadeau"];
                }
                if (!Array.isArray(g.stores)) {
                    g.stores = [];
                }
                if (typeof g.reservedBy !== 'string') {
                    g.reservedBy = g.reservedBy ? String(g.reservedBy) : "";
                }
                // Si l'item vient d'une ancienne version sans champ "order", on garde son ordre d'apparition actuel
                if (typeof g.order !== 'number' || isNaN(g.order)) {
                    g.order = idx + 1;
                }
            });

            gifts.sort((a, b) => {
                const orderA = (typeof a.order === 'number' && !isNaN(a.order)) ? a.order : 999999;
                const orderB = (typeof b.order === 'number' && !isNaN(b.order)) ? b.order : 999999;
                if (orderA !== orderB) return orderA - orderB;
                return (a.createdAt || 0) - (b.createdAt || 0);
            });

            gifts.forEach((g, idx) => {
                g.order = idx + 1;
            });
        }

        // Récupération des identifiants JSONBin actifs (Priorité au code JSONBIN_CONFIG, sinon configuration Admin locale)
        function getActiveJsonbinCredentials() {
            let saved = { BIN_ID: "", API_KEY: "" };
            try {
                const raw = localStorage.getItem(JSONBIN_LOCAL_CFG_KEY) || localStorage.getItem(JSONBIN_ALT_CFG_KEY);
                if (raw) {
                    const parsed = JSON.parse(raw);
                    saved.BIN_ID = parsed.BIN_ID || parsed.binId || "";
                    saved.API_KEY = parsed.API_KEY || parsed.apiKey || "";
                }
            } catch (e) {}

            return {
                binId: (JSONBIN_CONFIG.BIN_ID || saved.BIN_ID || "").trim(),
                apiKey: (JSONBIN_CONFIG.API_KEY || saved.API_KEY || "").trim()
            };
        }

        function saveJsonbinCredentialsLocally(binId, apiKey) {
            try {
                const payload = JSON.stringify({ BIN_ID: binId, API_KEY: apiKey, binId, apiKey });
                localStorage.setItem(JSONBIN_LOCAL_CFG_KEY, payload);
                localStorage.setItem(JSONBIN_ALT_CFG_KEY, payload);
            } catch (e) {}
        }

        function isJsonbinConfigured() {
            const { binId, apiKey } = getActiveJsonbinCredentials();
            return Boolean(binId && apiKey);
        }

        // Construit les en-têtes d'authentification compatibles avec X-Master-Key ou X-Access-Key
        function buildJsonbinHeaders(apiKey, includeContentType = false) {
            const headers = {
                'X-Bin-Meta': 'false'
            };
            if (includeContentType) {
                headers['Content-Type'] = 'application/json';
            }
            if (apiKey.startsWith('$2b$')) {
                headers['X-Access-Key'] = apiKey;
            } else {
                headers['X-Master-Key'] = apiKey;
            }
            return headers;
        }

        // Extrait la liste de cadeaux quelle que soit la structure enregistrée dans JSONBin
        function extractGiftsFromRecord(record) {
            if (!record) return null;
            if (Array.isArray(record)) return record;
            if (Array.isArray(record.gifts)) return record.gifts;
            if (Array.isArray(record.items)) return record.items;
            if (record.record) return extractGiftsFromRecord(record.record);
            return null;
        }

        // Mise à jour visuelle du badge d'état dans l'en-tête
        function updateSyncStatusBadge(state = 'auto') {
            const badge = document.getElementById('syncBadge');
            if (!badge) return;

            if (state === 'syncing') {
                badge.className = "inline-flex items-center gap-1.5 text-[11px] font-semibold bg-sky-50 text-sky-700 border border-sky-200 px-2.5 py-0.5 rounded-full";
                badge.innerHTML = `☁️ Synchronisation...`;
                return;

            if (state === 'error') {

                return;

        // Lecture & Écriture LocalStorage (Fallback immédiat)
        function loadFromLocalStorage() {
            try {
                const storedPin = localStorage.getItem(PIN_STORAGE_KEY);
                if (storedPin) adminPin = storedPin;

                const raw = localStorage.getItem(STORAGE_KEY);
                const alreadySeeded = localStorage.getItem(SEEDED_KEY) === 'true';
                if (raw !== null) {
                    const parsed = JSON.parse(raw);
                    if (Array.isArray(parsed)) return parsed;
                }
                if (!alreadySeeded) {
                    localStorage.setItem(STORAGE_KEY, JSON.stringify(DEFAULT_GIFTS));
                    localStorage.setItem(SEEDED_KEY, 'true');
                    return [...DEFAULT_GIFTS];
                }
                return [];
            } catch (e) {
                return [...DEFAULT_GIFTS];
            }
        }

        function saveToLocalStorage(giftsArray) {
            try {
                localStorage.setItem(STORAGE_KEY, JSON.stringify(giftsArray));
                localStorage.setItem(SEEDED_KEY, 'true');
                localStorage.setItem(PIN_STORAGE_KEY, adminPin);
            } catch (e) {}
        }

        // 1. Lecture depuis JSONBin.io (GET /v3/b/${BIN_ID}/latest) - NE MODIFIE JAMAIS LE BIN DISTANT
        async function fetchFromJsonbin(silent = false) {
            const { binId, apiKey } = getActiveJsonbinCredentials();
            if (!binId || !apiKey) {
                updateSyncStatusBadge();
                return null;
            }

            if (!silent) updateSyncStatusBadge('syncing');
            try {
                const response = await fetch(`https://api.jsonbin.io/v3/b/${encodeURIComponent(binId)}/latest`, {
                    method: 'GET',
                    headers: buildJsonbinHeaders(apiKey, false),
                    cache: 'no-store'
                });

                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}`);
                }

                const data = await response.json();
                const record = data.record !== undefined ? data.record : data;

                // Supporte tous les formats de sauvegarde JSONBin ({ gifts: [...] }, { items: [...] } ou [...])
                const remoteGifts = extractGiftsFromRecord(record);
                if (record && typeof record === 'object' && record.adminPin) {
                    adminPin = String(record.adminPin);
                }

                if (remoteGifts) {
                    gifts = remoteGifts;
                    normalizeGiftOrders();
                    saveToLocalStorage(gifts);
                    updateCategoryOptions();
                    renderGifts();
                    updateSyncStatusBadge('connected');
                    return gifts;
                }
                updateSyncStatusBadge('connected');
                return null;
            } catch (err) {
                console.warn("Erreur lors de la lecture JSONBin.io :", err);
                updateSyncStatusBadge('error');
                return null;
            }
        }

        // 2. Écriture vers JSONBin.io (PUT /v3/b/${BIN_ID})
        async function pushToJsonbin() {
            saveToLocalStorage(gifts);
            const { binId, apiKey } = getActiveJsonbinCredentials();
            if (!binId || !apiKey) {
                updateSyncStatusBadge();
                return false;
            }

            updateSyncStatusBadge('syncing');
            try {
                const payload = {
                    updatedAt: Date.now(),
                    adminPin: adminPin,
                    gifts: gifts
                };

                const response = await fetch(`https://api.jsonbin.io/v3/b/${encodeURIComponent(binId)}`, {
                    method: 'PUT',
                    headers: buildJsonbinHeaders(apiKey, true),
                    body: JSON.stringify(payload)
                });

                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}`);
                }

                updateSyncStatusBadge('connected');
                return true;
            } catch (err) {
                console.warn("Erreur lors de la sauvegarde sur JSONBin.io :", err);
                updateSyncStatusBadge('error');
                return false;
            }
        }

        // 3. Création automatique d'un nouveau Bin (POST /v3/b) avec uniquement la clé API
        window.createNewJsonbinAutomatically = async function() {
            const apiKeyInput = document.getElementById('jsonbinApiKeyInput').value.trim();
            if (!apiKeyInput) {
                showToast("⚠️ Veuillez d'abord coller votre Clé API JSONBin.io dans le champ prévu.");
                document.getElementById('jsonbinApiKeyInput').focus();
                return;
            }

            const btn = document.getElementById('createBinBtn');
            const originalText = btn.textContent;
            btn.disabled = true;
            btn.textContent = "⏳ Création du Bin en cours...";

            try {
                const payload = {
                    updatedAt: Date.now(),
                    adminPin: adminPin,
                    gifts: gifts
                };

                const headers = buildJsonbinHeaders(apiKeyInput, true);
                headers['X-Bin-Name'] = 'Ma_Liste_De_Noel';

                const response = await fetch('https://api.jsonbin.io/v3/b', {
                    method: 'POST',
                    headers: headers,
                    body: JSON.stringify(payload)
                });

                if (!response.ok) {
                    throw new Error(`Erreur API (${response.status})`);
                }

                const result = await response.json();
                const newBinId = result.metadata && result.metadata.id ? result.metadata.id : "";

                if (!newBinId) {
                    throw new Error("ID du Bin introuvable dans la réponse");
                }

                document.getElementById('jsonbinBinIdInput').value = newBinId;
                saveJsonbinCredentialsLocally(newBinId, apiKeyInput);

                updateConfigSnippetPreview();
                updateSyncStatusBadge('connected');
                showToast("🎉 Bin créé et connecté avec succès ! Copiez maintenant le bloc JSONBIN_CONFIG.");
            } catch (err) {
                showToast("❌ Impossible de créer le Bin. Vérifiez votre clé X-Master-Key.");
            } finally {
                btn.disabled = false;
                btn.textContent = originalText;
            }
        };

        // Bouton d'actualisation manuelle dans l'en-tête
        window.manualRefreshCloud = async function() {
            if (!isJsonbinConfigured()) {
                gifts = loadFromLocalStorage();
                updateCategoryOptions();
                renderGifts();
                showToast("💾 Liste locale rechargée (Configurez JSONBin dans Admin pour le Cloud)");
                return;
            }
            const res = await fetchFromJsonbin(false);
            if (res) {
                showToast("☁️ Liste synchronisée avec succès depuis JSONBin.io !");
            } else {
                showToast("⚠️ Impossible de joindre JSONBin.io, affichage de la version locale.");
            }
        };

        // Affichage de notifications Toast
        function showToast(message) {
            const toast = document.getElementById('toast');
            document.getElementById('toastText').textContent = message;
            toast.classList.remove('hidden');
            clearTimeout(window._toastTimer);
            window._toastTimer = setTimeout(() => {
                toast.classList.add('hidden');
            }, 3400);
        }

        // Initialisation au démarrage de la page
        async function initAppBackend() {
            // 1. Restaurer la session Admin si déjà déverrouillée
            try {
                if (sessionStorage.getItem(ADMIN_SESSION_KEY) === 'true' || localStorage.getItem(ADMIN_SESSION_KEY) === 'true') {
                    isAdmin = true;
                    updateAdminUI();
                }
            } catch (e) {}

            // 2. Si JSONBin.io est configuré, on récupère EN PRIORITÉ la liste en ligne sans jamais l'écraser
            if (isJsonbinConfigured()) {
                const remoteData = await fetchFromJsonbin(false);
                if (remoteData) {
                    document.getElementById('loadingState').classList.add('hidden');
                    return;
                }
            }

            // 3. Sinon (ou hors-ligne), affichage depuis le cache local
            gifts = loadFromLocalStorage();
            normalizeGiftOrders();
            document.getElementById('loadingState').classList.add('hidden');
            updateCategoryOptions();
            renderGifts();
            updateSyncStatusBadge();
        }

        // Sauvegarde d'un cadeau (Ajout ou Modification) + Envoi immédiat vers JSONBin.io
        async function persistGift(giftObj) {
            giftObj.updatedAt = Date.now();
            const idx = gifts.findIndex(g => g.id === giftObj.id);
            if (idx >= 0) gifts[idx] = giftObj;
            else gifts.push(giftObj);

            saveToLocalStorage(gifts);
            updateCategoryOptions();
            renderGifts();
            await pushToJsonbin();
        }

        // Suppression d'un cadeau + Envoi immédiat vers JSONBin.io
        async function removeGiftById(giftId) {
            gifts = gifts.filter(g => g.id !== giftId);
            normalizeGiftOrders();
            saveToLocalStorage(gifts);
            updateCategoryOptions();
            renderGifts();
            await pushToJsonbin();

        }

        // Déplacer un cadeau à une nouvelle position (1 à N) dans le classement par défaut
        window.moveGiftToPosition = async function(giftId, targetPos) {
            normalizeGiftOrders();
            const currentIndex = gifts.findIndex(g => g.id === giftId);
            if (currentIndex === -1) return;

            const newIndex = Math.max(0, Math.min(gifts.length - 1, Number(targetPos) - 1));
            if (currentIndex === newIndex) return;

            const [movedItem] = gifts.splice(currentIndex, 1);
            gifts.splice(newIndex, 0, movedItem);
            gifts.forEach((g, idx) => {
                g.order = idx + 1;
            });

            // Si un tri par prix était actif, repasser sur le tri personnalisé pour visualiser le changement
            const sortSelect = document.getElementById('sortFilter');
            if (sortSelect && sortSelect.value !== 'default') {
                sortSelect.value = 'default';
            }

            saveToLocalStorage(gifts);
            renderGifts();
            await pushToJsonbin();
            showToast(`↕️ Classement mis à jour (#${newIndex + 1})`);
        };

        // Déplacer un cadeau d'un cran vers le haut (-1) ou vers le bas (+1)
        window.shiftGiftOrder = async function(giftId, direction) {
            normalizeGiftOrders();
            const currentIndex = gifts.findIndex(g => g.id === giftId);
            if (currentIndex === -1) return;
            const targetPos = currentIndex + 1 + direction;
            if (targetPos < 1 || targetPos > gifts.length) return;
            await window.moveGiftToPosition(giftId, targetPos);
        };

            if (!gift.stores || gift.stores.length === 0) return 0;

        }

        window.submitReservation = async function(e) {
            e.preventDefault();
            const giftId = document.getElementById('reserveGiftId').value;
            const name = document.getElementById('reserverNameInput').value.trim();
            if (!name) return;

            const btn = document.getElementById('confirmReserveBtn');
            btn.disabled = true;

            try {
                // 1. Si JSONBin est actif, on récupère d'abord la dernière version Cloud pour éviter les doublons
                if (isJsonbinConfigured()) {
                    await fetchFromJsonbin(true);
                }

                const gift = gifts.find(g => g.id === giftId);
                if (!gift) {
                    closeReserveModal();
                    return;
                }

                // Vérifier si quelqu'un d'autre vient tout juste de le réserver
                if (gift.reservedBy && gift.reservedBy.trim() !== "") {
                    closeReserveModal();
                    renderGifts();
                    showToast(`⚠️ Ce cadeau vient d'être réservé par ${gift.reservedBy} !`);
                    return;
                }

                const updatedGift = {
                    ...gift,
                    reservedBy: name,
                    reservedAt: Date.now()
                };

                closeReserveModal();
                await persistGift(updatedGift);
                showToast(`🎁 Merci ${name} ! Le cadeau est réservé.`);
            } finally {
                btn.disabled = false;
            }
        };
